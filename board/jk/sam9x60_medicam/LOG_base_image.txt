            deb https://deb.debian.org/debian bookworm-updates main non-free-firmware
        - sudo apt update

1.2 [ ] (current: WSL2 Debian) isolate the Linux PATH from the Windows PATH
        why: Windows PATH is appended to the WSL PATH (/mnt/c/WINDOWS/system32,
             .../Git/usr/bin, .../Go/bin ...), so Windows make/gcc/find/python shadow
             the Linux ones and the build fails in confusing ways.
        sudo tee /etc/wsl.conf >/dev/null <<'EOF'
        [interop]
        appendWindowsPath = false
        EOF
        # from PowerShell: wsl --shutdown   then reopen the Debian shell
        verify: printenv PATH | tr ':' '\n' | grep -c /mnt/c      -> 0
        verify: which make gcc python3                            -> /usr/bin/*
        note:   still reach the Windows tree via /mnt/d/Source/...

1.3 [ ] apt packages
        sudo apt install -y git gitk meld which sed make binutils build-essential \
             diffutils gcc g++ bash patch gzip bzip2 perl tar cpio unzip rsync file \
             bc findutils libssl-dev libncurses-dev wget ca-certificates
        optional: gh (https://github.com/cli/cli/blob/trunk/docs/install_linux.md)
        note: mkpasswd is NOT installed by the above; use openssl passwd -6 for hashes.

1.4 [x] git identity
        git config --global user.name "Yuriy Kapoyko"
        git config --global user.email "yury.kapoyko@gmail.com"

1.5 [ ] host quirk when driving WSL from the Windows shell
        - MSYS/git-bash rewrites unix-looking arguments (/etc/x -> C:/Program Files/Git/etc/x)
        - always use:  wsl -d Debian -- bash -lc '<command>'
        - or prefix the single command with MSYS_NO_PATHCONV=1
        - symptom of a violation: a Linux command reporting a C:/... path, or a tool call
          that hangs waiting on stdin.

        evtest                                            # press the user button
        gpiodetect; gpioinfo
        gpioset gpiochip0 31=1                            # PA31 LED on
        gpioget gpiochip0 30                              # PA30 button

5.3 [x] I2C
        i2cdetect -l
        i2cdetect -y 0                                    # 0x50 EEPROM, 0x30 OV2640
        i2cdump -y 0 0x50
        dmesg | grep at24 ; ls /sys/bus/i2c/devices/0-0050/eeprom
        dd if=/sys/bus/i2c/devices/0-0050/eeprom bs=1 count=32768 | hexdump -C
        echo -n -e '\xAA' | dd of=/sys/bus/i2c/devices/0-0050/eeprom bs=1 count=1 seek=0

5.4 [x] SPI / WS2812
        NOTE: spi-tools 1.0.2 installs only spi-config and spi-pipe (there is no
              spi_xfer), and spidev_test is not in the image.
        printf '\xFF\x00\x00' | spi-pipe -d /dev/spidev1.0 -s 3200000   # WS2812 green
        printf '\x00\xFF\x00' | spi-pipe -d /dev/spidev1.0 -s 3200000   # WS2812 red
        spi-config -d /dev/spidev1.0 -q                                  # query mode/bits/speed

5.5 [x] USART / RS485
        dmesg | grep tty                                  # ttyS2, ttyS5
        stty -F /dev/ttyS2 115200; echo Hello > /dev/ttyS2; cat /dev/ttyS2
        screen /dev/ttyS2 115200
        stty -F /dev/ttyS5 115200
        gpioset gpiochip0 29=1 23=0 ; echo "Hello RS485" > /dev/ttyS5 ; sleep 0.1
        gpioset gpiochip0 29=0 23=1

5.6 [x] camera OV2640
        i2cdetect -y 0                                    # expect 0x30
        v4l2grab -d /dev/video0 -o test.jpg -W 320 -H 240 -q 85
        v4l2grab -d /dev/video0 -o full.jpg -W 1600 -H 1200
        NOTE: v4l2-ctl / media-ctl / fswebcam come from v4l-utils & friends, which are
              NOT in the trimmed image; only v4l2grab is kept (stage 1 decision).
